ld-graph

A dependency-free graph element. One attribute for data, one for type, everything else is CSS variables.

type=smooth

<graph ld-graph-data="1, 3, 5, 8, 10, 6, 9"
       ld-graph-type="smooth"></graph>

type=straight

<graph ld-graph-data="1, 3, 5, 8, 10, 6, 9"
       ld-graph-type="straight"></graph>

type=bar

<graph ld-graph-data="4, 8, 6, 10, 5, 9, 7"
       ld-graph-type="bar"></graph>

type=dots

<graph ld-graph-data="2, 6, 4, 9, 3, 8, 5"
       ld-graph-type="dots"></graph>

type=step

<graph ld-graph-data="3, 3, 6, 6, 2, 2, 8"
       ld-graph-type="step"></graph>

ld-graph-area

<graph ld-graph-data="1, 4, 3, 7, 5, 9, 6"
       ld-graph-type="smooth"
       ld-graph-area></graph>

ld-graph-points

<graph ld-graph-data="5, 2, 8, 4, 7, 3, 9"
       ld-graph-type="straight"
       ld-graph-points></graph>

ld-graph-animate

<graph ld-graph-data="2, 5, 3, 8, 6, 9, 4"
       ld-graph-type="bar"
       ld-graph-animate></graph>

comparison — smooth + area + legend

<graph
  ld-graph-data="12,19,14,22,18,25,21; 8,11,15,12,17,14,19"
  ld-graph-type="smooth"
  ld-graph-area
  ld-graph-labels="Mon, Tue, Wed, Thu, Fri, Sat, Sun"
  ld-graph-series-labels="This week, Last week"
  ld-graph-legend
></graph>

comparison — grouped bar

<graph
  ld-graph-data="4,8,6,10,5; 6,5,9,7,8; 3,6,4,8,6"
  ld-graph-type="bar"
  ld-graph-series-labels="A, B, C"
  ld-graph-legend></graph>

comparison — stacked bar

<graph
  ld-graph-data="4,8,6,10,5; 6,5,9,7,8; 3,6,4,8,6"
  ld-graph-type="bar"
  ld-graph-stack
  ld-graph-series-labels="A, B, C"
  ld-graph-legend></graph>

ld-graph-labels + custom colors

<graph
  ld-graph-data="12, 19, 14, 22, 18, 25, 21"
  ld-graph-type="smooth"
  ld-graph-area
  ld-graph-labels="Mon, Tue, Wed, Thu, Fri, Sat, Sun"
  style="--ld-graph-color: #ff6bd6;"
></graph>
